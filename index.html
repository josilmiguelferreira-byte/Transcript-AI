const express = require('express'); const cors = require('cors'); const multer = require('multer'); const bcrypt = require('bcryptjs'); const jwt = require('jsonwebtoken'); const { AssemblyAI } = require('assemblyai'); const { OAuth2Client } = require('google-auth-library'); const Stripe = require('stripe'); const PocketBase = require('pocketbase/cjs'); const fs = require('fs'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; // ───────────────────────────────────────────── // CONFIGURATIE — via Railway Environment Variables // ───────────────────────────────────────────── const JWT_SECRET = process.env.JWT_SECRET || 'verander-dit-naar-een-geheim'; const ASSEMBLYAI_API_KEY = process.env.ASSEMBLYAI_API_KEY || ''; const STRIPE_SECRET_KEY = process.env.STRIPE_SECRET_KEY || ''; const STRIPE_PRO_PRICE = process.env.STRIPE_PRO_PRICE_ID || ''; const GOOGLE_CLIENT_ID = process.env.GOOGLE_CLIENT_ID || ''; const POCKETBASE_URL = process.env.POCKETBASE_URL || 'http://127.0.0.1:8090'; const FRONTEND_URL = process.env.FRONTEND_URL || '*'; // ───────────────────────────────────────────── // CLIENTS // ───────────────────────────────────────────── const assemblyClient = ASSEMBLYAI_API_KEY ? new AssemblyAI({ apiKey: ASSEMBLYAI_API_KEY }) : null; const stripeClient = STRIPE_SECRET_KEY ? new Stripe(STRIPE_SECRET_KEY) : null; const googleClient = new OAuth2Client(GOOGLE_CLIENT_ID); const pb = new PocketBase(POCKETBASE_URL); // Multer — opslaan in /tmp (Railway heeft geen permanente opslag) const upload = multer({ dest: '/tmp/', limits: { fileSize: 500 * 1024 * 1024 } // 500MB max }); // ───────────────────────────────────────────── // MIDDLEWARE // ───────────────────────────────────────────── app.use(cors({ origin: [FRONTEND_URL, 'https://transcriptai-1hj.pages.dev', 'http://localhost', 'http://localhost:3000'], credentials: true, methods: ['GET', 'POST', 'PUT', 'DELETE', 'OPTIONS'], allowedHeaders: ['Content-Type', 'Authorization'] })); app.use(express.json()); app.use(express.urlencoded({ extended: true })); // Auth middleware — controleert JWT token function requireAuth(req, res, next) { const authHeader = req.headers['authorization']; const token = authHeader && authHeader.split(' ')[1]; if (!token) return res.status(401).json({ error: 'Not authenticated' }); try { const decoded = jwt.verify(token, JWT_SECRET); req.userId = decoded.userId; req.userEmail = decoded.email; next(); } catch (e) { return res.status(401).json({ error: 'Invalid or expired token' }); } } // Helper: maak JWT token aan function makeToken(user) { return jwt.sign( { userId: user.id, email: user.email }, JWT_SECRET, { expiresIn: '7d' } ); } // Helper: haal user op uit PocketBase async function getUser(userId) { try { return await pb.collection('users').getOne(userId); } catch (e) { return null; } } // Helper: PocketBase admin auth voor schrijven naar collections const PB_ADMIN_EMAIL = process.env.PB_ADMIN_EMAIL || ''; const PB_ADMIN_PASSWORD = process.env.PB_ADMIN_PASSWORD || ''; async function pbAdminAuth() { try { if (PB_ADMIN_EMAIL && PB_ADMIN_PASSWORD && !pb.authStore.isValid) { await pb.collection('_superusers').authWithPassword(PB_ADMIN_EMAIL, PB_ADMIN_PASSWORD); } } catch (e) { console.warn('PocketBase admin auth failed:', e.message); } } // ───────────────────────────────────────────── // HEALTH CHECK // ───────────────────────────────────────────── app.get('/api/health', (req, res) => { res.json({ status: 'ok', timestamp: new Date().toISOString(), services: { assemblyai: !!assemblyClient, stripe: !!stripeClient, pocketbase: POCKETBASE_URL } }); }); // ───────────────────────────────────────────── // AUTH ROUTES // ───────────────────────────────────────────── // Registreren app.post('/api/auth/register', async (req, res) => { try { const { email, password, name } = req.body; if (!email || !password) return res.status(400).json({ error: 'Email and password required' }); if (password.length < 8) return res.status(400).json({ error: 'Password must be at least 8 characters' }); // FIX: Check of email al bestaat try { const existing = await pb.collection('users').getFirstListItem(`email="${email}"`); if (existing) return res.status(400).json({ error: 'An account with this email already exists. Please log in instead.' }); } catch (e) { // Geen user gevonden — goed, we kunnen doorgaan } // Maak user aan in PocketBase const user = await pb.collection('users').create({ email, password, passwordConfirm: password, name: name || email.split('@')[0], isPro: false, isBusiness: false, minutesUsed: 0 }); const token = makeToken(user); res.json({ token, user: { id: user.id, email: user.email, name: user.name, isPro: user.isPro || false, isBusiness: user.isBusiness || false, minutesUsed: user.minutesUsed || 0 } }); } catch (e) { console.error('Register error:', e); const msg = e?.response?.data?.email?.message || e.message || 'Registration failed'; res.status(400).json({ error: msg }); } }); // Inloggen app.post('/api/auth/login', async (req, res) => { try { const { email, password } = req.body; if (!email || !password) return res.status(400).json({ error: 'Email and password required' }); const authData = await pb.collection('users').authWithPassword(email, password); const token = makeToken(authData.record); res.json({ token, user: { id: authData.record.id, email: authData.record.email, name: authData.record.name, isPro: authData.record.isPro, isBusiness: authData.record.isBusiness, minutesUsed: authData.record.minutesUsed } }); } catch (e) { console.error('Login error:', e); res.status(401).json({ error: 'Invalid email or password' }); } }); // Uitloggen app.post('/api/auth/logout', (req, res) => { res.json({ success: true }); }); // Huidige gebruiker ophalen app.get('/api/auth/me', requireAuth, async (req, res) => { try { const user = await getUser(req.userId); if (!user) return res.status(404).json({ error: 'User not found' }); res.json({ id: user.id, email: user.email, name: user.name, isPro: user.isPro, isBusiness: user.isBusiness, minutesUsed: user.minutesUsed }); } catch (e) { res.status(401).json({ error: 'Could not fetch user' }); } }); // Google OAuth app.post('/api/auth/google', async (req, res) => { try { const { credential } = req.body; if (!credential) return res.status(400).json({ error: 'No credential provided' }); // Verifieer Google token const ticket = await googleClient.verifyIdToken({ idToken: credential, audience: GOOGLE_CLIENT_ID }); const payload = ticket.getPayload(); const { email, name, sub: googleId } = payload; // Check of user al bestaat let user; try { const results = await pb.collection('users').getList(1, 1, { filter: `email = "${email}"` }); if (results.items.length > 0) { user = results.items[0]; } } catch (e) { /* user bestaat niet */ } // Maak user aan als die niet bestaat if (!user) { user = await pb.collection('users').create({ email, password: googleId + JWT_SECRET, passwordConfirm: googleId + JWT_SECRET, name: name || email.split('@')[0], googleId, isPro: false, isBusiness: false, minutesUsed: 0 }); } const token = makeToken(user); res.json({ token, user: { id: user.id, email: user.email, name: user.name, isPro: user.isPro, isBusiness: user.isBusiness, minutesUsed: user.minutesUsed } }); } catch (e) { console.error('Google auth error:', e); res.status(401).json({ error: 'Google authentication failed: ' + e.message }); } }); // ───────────────────────────────────────────── // TRANSCRIPTIE ROUTES // ───────────────────────────────────────────── // Upload en transcribeer bestand app.post('/api/transcription/upload', requireAuth, upload.single('file'), async (req, res) => { const filePath = req.file?.path; try { if (!req.file) return res.status(400).json({ error: 'No file uploaded' }); if (!assemblyClient) return res.status(503).json({ error: 'AssemblyAI not configured — add ASSEMBLYAI_API_KEY in Railway Variables' }); const user = await getUser(req.userId); if (!user) return res.status(401).json({ error: 'User not found' }); const mode = req.body.mode || 'standard'; const language = req.body.language || 'auto'; // Bouw AssemblyAI opties const transcriptOptions = { audio: fs.createReadStream(filePath), speaker_labels: mode === 'speakers', auto_highlights: false, punctuate: true, format_text: true, }; // Taal instellen if (language !== 'auto') { transcriptOptions.language_code = language; } else { transcriptOptions.language_detection = true; } // Summary mode if (mode === 'summary') { transcriptOptions.summarization = true; transcriptOptions.summary_model = 'informative'; transcriptOptions.summary_type = 'bullets'; } // Stuur naar AssemblyAI const transcript = await assemblyClient.transcripts.transcribe(transcriptOptions); if (transcript.status === 'error') { throw new Error(transcript.error || 'Transcription failed'); } // Bereken duur in minuten const durationMs = transcript.audio_duration || 0; const durationMin = Math.ceil(durationMs / 60); // Update minutesUsed voor gratis gebruikers let newMinutesUsed = user.minutesUsed || 0; if (!user.isPro && !user.isBusiness) { newMinutesUsed = newMinutesUsed + durationMin; await pb.collection('users').update(user.id, { minutesUsed: newMinutesUsed }); } // Sla op in geschiedenis try { await pbAdminAuth(); await pb.collection('transcriptions').create({ userId: user.id, filename: req.file.originalname, text: transcript.text || '', language: transcript.language_code || language, duration: durationMin, mode }); } catch (e) { console.warn('Could not save to history:', e.message); } // Verwijder tijdelijk bestand if (filePath && fs.existsSync(filePath)) fs.unlinkSync(filePath); res.json({ transcript: { text: transcript.text, words: transcript.words || [], utterances: transcript.utterances || [], summary: transcript.summary || null, language_code: transcript.language_code || language, audio_duration: durationMs }, minutesUsed: newMinutesUsed }); } catch (e) { // Verwijder tijdelijk bestand bij fout if (filePath && fs.existsSync(filePath)) { try { fs.unlinkSync(filePath); } catch (_) {} } console.error('Transcription error:', e); res.status(500).json({ error: e.message || 'Transcription failed' }); } }); // Geschiedenis ophalen app.get('/api/transcription/history', requireAuth, async (req, res) => { try { const results = await pb.collection('transcriptions').getList(1, 50, { filter: `userId = "${req.userId}"`, sort: '-created' }); res.json({ history: results.items.map(item => ({ id: item.id, filename: item.filename, text: item.text, language: item.language, duration: item.duration, mode: item.mode, createdAt: item.created })) }); } catch (e) { console.error('History error:', e); res.json({ history: [] }); } }); // ───────────────────────────────────────────── // PAYMENT ROUTES // ───────────────────────────────────────────── // Maak Stripe abonnement aan app.post('/api/payments/create-subscription', requireAuth, async (req, res) => { try { if (!stripeClient) return res.status(503).json({ error: 'Stripe not configured — add STRIPE_SECRET_KEY in Railway Variables' }); const { paymentMethodId, plan } = req.body; const user = await getUser(req.userId); if (!user) return res.status(404).json({ error: 'User not found' }); const priceId = plan === 'business' ? process.env.STRIPE_BUSINESS_PRICE_ID : STRIPE_PRO_PRICE; if (!priceId) return res.status(400).json({ error: 'Price ID not configured in Railway Variables' }); // Maak of haal Stripe customer op let customerId = user.stripeCustomerId; if (!customerId) { const customer = await stripeClient.customers.create({ email: user.email, name: user.name, payment_method: paymentMethodId, invoice_settings: { default_payment_method: paymentMethodId } }); customerId = customer.id; await pb.collection('users').update(user.id, { stripeCustomerId: customerId }); } else { await stripeClient.paymentMethods.attach(paymentMethodId, { customer: customerId }); await stripeClient.customers.update(customerId, { invoice_settings: { default_payment_method: paymentMethodId } }); } // Maak subscription aan met 7-daagse trial const subscription = await stripeClient.subscriptions.create({ customer: customerId, items: [{ price: priceId }], trial_period_days: 7, payment_settings: { payment_method_types: ['card'], save_default_payment_method: 'on_subscription' }, expand: ['latest_invoice.payment_intent'] }); const invoice = subscription.latest_invoice; const paymentIntent = invoice?.payment_intent; res.json({ subscriptionId: subscription.id, clientSecret: paymentIntent?.client_secret || null, status: subscription.status }); } catch (e) { console.error('Payment error:', e); res.status(400).json({ error: e.message || 'Payment failed' }); } }); // Bevestig betaling en activeer Pro app.post('/api/payments/confirm', requireAuth, async (req, res) => { try { const { subscriptionId } = req.body; if (!subscriptionId) return res.status(400).json({ error: 'No subscription ID' }); if (stripeClient) { const subscription = await stripeClient.subscriptions.retrieve(subscriptionId); const isPro = subscription.status === 'active' || subscription.status === 'trialing'; const isBusiness = subscription.items?.data?.[0]?.price?.id === process.env.STRIPE_BUSINESS_PRICE_ID; await pb.collection('users').update(req.userId, { isPro: isPro && !isBusiness, isBusiness, subscriptionId }); return res.json({ success: true, isPro, isBusiness }); } // Fallback als Stripe niet geconfigureerd is await pb.collection('users').update(req.userId, { isPro: true }); res.json({ success: true, isPro: true }); } catch (e) { console.error('Confirm error:', e); res.status(400).json({ error: e.message }); } }); // Stripe Webhook (voor automatische verlenging/cancellatie) app.post('/api/payments/webhook', express.raw({ type: 'application/json' }), async (req, res) => { if (!stripeClient) return res.json({ received: true }); const sig = req.headers['stripe-signature']; const secret = process.env.STRIPE_WEBHOOK_SECRET; let event; try { event = secret ? stripeClient.webhooks.constructEvent(req.body, sig, secret) : JSON.parse(req.body); } catch (e) { return res.status(400).json({ error: 'Webhook signature failed' }); } try { switch (event.type) { case 'customer.subscription.deleted': { const sub = event.data.object; const results = await pb.collection('users').getList(1, 1, { filter: `stripeCustomerId = "${sub.customer}"` }); if (results.items[0]) { await pb.collection('users').update(results.items[0].id, { isPro: false, isBusiness: false, subscriptionId: '' }); } break; } case 'invoice.payment_failed': { console.log('Payment failed for customer:', event.data.object.customer); break; } } } catch (e) { console.error('Webhook handler error:', e); } res.json({ received: true }); }); // ───────────────────────────────────────────── // START SERVER // ───────────────────────────────────────────── app.listen(PORT, () => { console.log(`✅ TranscriptAI backend running on port ${PORT}`); console.log(` AssemblyAI: ${assemblyClient ? '✓ configured' : '✗ missing ASSEMBLYAI_API_KEY'}`); console.log(` Stripe: ${stripeClient ? '✓ configured' : '✗ missing STRIPE_SECRET_KEY'}`); console.log(` PocketBase: ${POCKETBASE_URL}`); console.log(` Google: ${GOOGLE_CLIENT_ID ? '✓ configured' : '✗ missing GOOGLE_CLIENT_ID'}`); });
